    @Test void oaLeaveDateAndReasonRules() {
        for(String reason:new String[]{null," ","abcd","abcde","x".repeat(300),"x".repeat(301),"<valid","valid>","12345","12a45"}) {
            LeaveAssessment a=oaAssess(oaLeave(LeaveType.SICK,EmployeeCategory.REGULAR,LeaveUnit.FULL_DAY,0,OA_DAY,OA_DAY,reason,false,false,true,12,7,50,0));
            if(reason==null||reason.isBlank()){assertEquals(List.of("reason is required"),a.violations());continue;}
            assertEquals(reason.length()<5,a.violations().contains("reason must contain at least five characters"));assertEquals(reason.length()>300,a.violations().contains("reason cannot exceed 300 characters"));assertEquals(reason.contains("<")||reason.contains(">"),a.violations().contains("reason contains unsupported markup"));assertEquals(reason.equals("12345"),a.violations().contains("reason cannot contain digits only"));
        }
        AdvancedLeaveRequest r=oaDays(LeaveType.SICK,EmployeeCategory.REGULAR,1,false,false,true,12,7,50,0);LeaveAssessment a=new AdvancedLeavePolicy().assess(r,d->false,OA_DAY);assertEquals(List.of("request contains no working time"),a.violations());
        for(LeaveType type:LeaveType.values())for(boolean emergency:new boolean[]{false,true})for(int offset:new int[]{-1,0,365,366}) {
            r=oaLeave(type,EmployeeCategory.REGULAR,LeaveUnit.FULL_DAY,0,OA_DAY.plusDays(offset),OA_DAY.plusDays(offset),"valid",emergency,false,true,12,7,50,0);
            oaViolation(r,"backdated leave is only allowed for emergency sick leave",offset<0&&!(type==LeaveType.SICK&&emergency));oaViolation(r,"leave cannot be requested more than one year ahead",offset>365);
        }
        for(int days:new int[]{1,2,5,6,10,11,45,46})for(int notice:new int[]{0,1,2,3,6,7})for(boolean emergency:new boolean[]{false,true}) {
            r=oaDays(LeaveType.SICK,EmployeeCategory.REGULAR,days,emergency,false,true,12,notice,50,0);
            oaViolation(r,"multi-day leave requires advance notice",notice==0&&days>1&&!emergency);
            oaViolation(r,"leave longer than five days requires three days notice",notice<3&&days>5&&!emergency);
            oaViolation(r,"leave longer than ten days requires seven days notice",notice<7&&days>10&&!emergency);
            oaViolation(r,"a single request cannot span more than 45 calendar days",days>45);
        }
    }
